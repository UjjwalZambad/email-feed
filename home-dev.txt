      (function () {
        var currentStateKey = "portalCurrentState";
        var firstVisitKey = "homeFirstVisitRedirect_cat_customer";
        var previousState = localStorage.getItem(currentStateKey);

        /*
          Redirect Customer when:
          1. Customer logs in and lands on Home, or
          2. Customer lands on Home for the first time in this browser
        */
        if (previousState !== "customer" || !localStorage.getItem(firstVisitKey)) {
          localStorage.setItem(currentStateKey, "customer");
          localStorage.setItem(firstVisitKey, "true");
          window.location.replace("{{ helpArticlesUrl }}");
          return;
        }

        localStorage.setItem(currentStateKey, "customer");
      })();
    </script>

      var previousState = localStorage.getItem(currentStateKey);

      /*
        Redirect Guest when:
        1. User has just logged out, meaning previous state was not guest, or
        2. Guest lands on Home for the first time in this browser
      */
      if (previousState && previousState !== "guest") {
        localStorage.setItem(currentStateKey, "guest");
        localStorage.setItem(guestFirstVisitKey, "true");
        window.location.replace("{{ helpArticlesUrl }}");
        return;

      if (!localStorage.getItem(guestFirstVisitKey)) {
        localStorage.setItem(currentStateKey, "guest");
        localStorage.setItem(guestFirstVisitKey, "true");
        window.location.replace("{{ helpArticlesUrl }}");
        return;
      }

      localStorage.setItem(currentStateKey, "guest");
    })();
