// Always resolves: returns null when the id is missing or the call fails
function fetchOptionalRecord(entitySet, id, select) {
    const d = $.Deferred();

    if (!id) {
        d.resolve(null);
        return d.promise();
    }

    $.ajax({
        url: "/_api/" + entitySet + "(" + id + ")?$select=" + select,
        method: "GET",
        headers: {
            "Accept": "application/json",
            "Prefer": 'odata.include-annotations="OData.Community.Display.V1.FormattedValue"'
        }
    })
    .done(function (res) { d.resolve(res || null); })
    .fail(function (xhr) {
        console.warn("Optional lookup failed:", entitySet, id, xhr && xhr.status);
        d.resolve(null);
    });

    return d.promise();
}

// Loads the incident, then contact / requested-by / parent case in parallel
function loadHandupData(recordId) {
    const d = $.Deferred();

    $.ajax({
        url: "/_api/incidents(" + recordId + ")?$select=" + HANDUP_INCIDENT_SELECT,
        method: "GET",
        headers: { "Accept": "application/json" }
    })
    .done(function (data) {
        $.when(
            fetchOptionalRecord("contacts", data._customerid_value,
                "fullname,firstname,lastname,emailaddress1,address1_telephone1"),
            fetchOptionalRecord("contacts", data._cat_requestedfrom_value,
                "fullname,firstname,lastname,emailaddress1,address1_telephone1,cat_persona"),
            fetchOptionalRecord("incidents", data._parentcaseid_value, "ticketnumber")
        ).done(function (contact, requestedBy, parentCase) {
            data.contact = contact;
            data.requestedBy = requestedBy;
            data.parentCaseTicketNumber = parentCase ? parentCase.ticketnumber : "";
            console.log("ElevationType:", data.cat_elevationtype);
            d.resolve(data);
        });
    })
    .fail(function (xhr) {
        console.error("Incident load failed:", xhr);
        d.reject(xhr);
    });

    return d.promise();
}

  let expandedRow = null;
  let cache = {};

  $(document).on("click", "#dealerCasesContainer table tbody tr", function () {

      if ($(this).hasClass("no-filter-results-row")) {
          return;
      }

      if (window.innerWidth < 768) {
          $("body").addClass("mobile-detail-view");
      }

      const row = $(this);
      const recordId = row.attr("data-id");

      // ---------- MOBILE ----------
      if (window.innerWidth <= 768) {

          $("#dealerCasesContainer").hide();
          $("#mobile-case-view")
              .show()
              .html("<div style='padding:20px'>Loading...</div>");

          loadHandupData(recordId)
              .done(function (data) {
                  $("#mobile-case-view").html(buildMobileCaseHtml(data, recordId));
              })
              .fail(function () {
                  $("#mobile-case-view").html(
                      "<div class='mobile-case-container'><div class='mobile-header'>" +
                      "<span class='back-btn'>←</span><h3>Handup</h3></div>" +
                      "<div style='padding:20px;color:#a00;'>Unable to load handup details.</div></div>"
                  );
              });

          return;
      }

      // ---------- DESKTOP ----------
      if (!recordId) return;

      // Toggle collapse
      if (expandedRow && expandedRow.is(row)) {
          row.next(".case-details").stop(true, true).slideUp(200, function () {
              $(this).remove();
          });
          expandedRow = null;
          return;
      }

      $(".case-details").stop(true, true).slideUp(200, function () {
          $(this).remove();
      });
      expandedRow = row;

      if (cache[recordId]) {
          renderRow(row, cache[recordId]);
          return;
      }

      // Loading skeleton
      row.after(`
  <tr class="case-details">
  <td colspan="10">
  <div style="padding:20px;">Loading...</div>
  </td>
  </tr>
      `);

      loadHandupData(recordId)
          .done(function (data) {
              cache[recordId] = data;
              row.next(".case-details").remove();
              renderRow(row, data);
          })
          .fail(function () {
              row.next(".case-details").html(
                  '<td colspan="10"><div style="padding:20px;color:#a00;">Unable to load handup details.</div></td>'
              );
              expandedRow = null;
          });
  });
